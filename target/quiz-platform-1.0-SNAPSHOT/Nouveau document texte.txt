🔑 Points de sécurité critiques

Point Méthode
Token API Ne jamais hardcoder → variables d'environnement
Webhook Vérifier signature HMAC + whitelist IP
Référence Unique, non devinable (UUID + timestamp)
Accès cours Vérifier paiement SUCCESS avant de débloquer
Rate limiting Max 5 tentatives/heure par utilisateur
Logs Ne jamais logguer token ou infos bancaires
HTTPS Obligatoire en production
Session Vérifier que l'utilisateur connecté = acheteur

---

✅ Validation avant passage en production

· Tous les endpoints sont en HTTPS
· Le webhook est accessible publiquement
· La signature HMAC est validée côté backend
· Les IP de PawaPay sont en whitelist
· Les montants sont validés (pas de paiement à 0 ou négatif)
· Les références sont uniques
· Rate limiting actif
· Logs sécurisés sans données sensibles
· Testé avec sandbox PawaPay
· Scripts SQL versionnés
· Rollback possible en cas d'échec

-- Table des paiements
CREATE TABLE IF NOT EXISTS payments (
    id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT NOT NULL,
    course_id INT NOT NULL,
    reference VARCHAR(100) UNIQUE NOT NULL,
    transaction_id VARCHAR(100),
    amount DECIMAL(10,2) NOT NULL,
    currency VARCHAR(3) DEFAULT 'USD',
    phone_number VARCHAR(20),
    status VARCHAR(20) DEFAULT 'PENDING',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_user_id (user_id),
    INDEX idx_course_id (course_id),
    INDEX idx_reference (reference),
    INDEX idx_status (status),
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    FOREIGN KEY (course_id) REFERENCES cours(id) ON DELETE CASCADE
);

-- Table des accès aux cours
CREATE TABLE IF NOT EXISTS user_course_access (
    id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT NOT NULL,
    course_id INT NOT NULL,
    granted_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    expires_at TIMESTAMP NULL,
    INDEX idx_user_course (user_id, course_id),
    UNIQUE KEY uk_user_course (user_id, course_id),
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    FOREIGN KEY (course_id) REFERENCES cours(id) ON DELETE CASCADE
);

// FICHIER : src/main/java/com/amijayquiz/controllers/PlanCoursServlet.java
// RÔLE    : Contrôleur du plan détaillé d'un cours
//           Affiche l'arborescence complète du cours :
//           tous les modules (1 à 15) avec chapitres, leçons,
//           examens et progression par module
//           Navigation vers les leçons via la page chapitre
// ============================================================

package com.amijayquiz.controllers;

import com.amijayquiz.dao.CertificatDAO;
import com.amijayquiz.dao.ChapitreDAO;
import com.amijayquiz.dao.CoursDAO;
import com.amijayquiz.dao.ExamenModuleDAO;
import com.amijayquiz.dao.LeconDAO;
import com.amijayquiz.dao.ProgressionCoursDAO;
import com.amijayquiz.dao.ProgressionModuleDAO;
import com.amijayquiz.models.Certificat;
import com.amijayquiz.models.Chapitre;
import com.amijayquiz.models.Cours;
import com.amijayquiz.models.ExamenModule;
import com.amijayquiz.models.Lecon;
import com.amijayquiz.models.ProgressionModule;

import javax.servlet.ServletException;
import javax.servlet.annotation.WebServlet;
import javax.servlet.http.HttpServlet;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;
import javax.servlet.http.HttpSession;
import java.io.IOException;
import java.util.ArrayList;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;

/**
 * PlanCoursServlet - Plan détaillé d'un cours organisé par modules
 * 
 * Affiche l'arborescence complète du cours regroupée par modules (1 à 15).
 * Chaque module contient :
 * - Ses chapitres avec leurs leçons
 * - Le statut de progression (verrouillé, débloqué, réussi)
 * - Un bouton pour passer l'examen du module
 * - Des liens pour accéder aux chapitres et leçons
 * 
 * URL : /joueur/cours/plan?coursId=1
 * 
 * @author AmijayQuiz
 * @version 2.1
 */
@WebServlet(name = "PlanCoursServlet", urlPatterns = {"/joueur/cours/plan"})
public class PlanCoursServlet extends HttpServlet {

    // Constantes des modules
    private static final int NOMBRE_MODULES = 15;

    @Override
    protected void doGet(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {

        // ============================================================
        // 1. VÉRIFICATION DE LA SESSION ET DU RÔLE
        // ============================================================
        HttpSession session = request.getSession(false);
        if (session == null || session.getAttribute("utilisateur") == null) {
            response.sendRedirect(request.getContextPath() + "/connexion");
            return;
        }

        Utilisateur utilisateur = (Utilisateur) session.getAttribute("utilisateur");
        if (!"joueur".equals(utilisateur.getRole())) {
            response.sendRedirect(request.getContextPath() + "/connexion");
            return;
        }

        int utilisateurId = utilisateur.getId();

        // ============================================================
        // 2. RÉCUPÉRATION ET VALIDATION DU PARAMÈTRE COURSID
        // ============================================================
        String coursIdParam = request.getParameter("coursId");
        if (coursIdParam == null || coursIdParam.trim().isEmpty()) {
            response.sendRedirect(request.getContextPath() + "/joueur/cours");
            return;
        }

        int coursId;
        try {
            coursId = Integer.parseInt(coursIdParam);
        } catch (NumberFormatException e) {
            response.sendRedirect(request.getContextPath() + "/joueur/cours");
            return;
        }

        // ============================================================
        // 3. INITIALISATION DES DAO
        // ============================================================
        CoursDAO coursDAO = new CoursDAO();
        ChapitreDAO chapitreDAO = new ChapitreDAO();
        LeconDAO leconDAO = new LeconDAO();
        ProgressionCoursDAO progressionCoursDAO = new ProgressionCoursDAO();
        ProgressionModuleDAO progressionModuleDAO = new ProgressionModuleDAO();
        ExamenModuleDAO examenModuleDAO = new ExamenModuleDAO();
        CertificatDAO certificatDAO = new CertificatDAO();

        // ============================================================
        // 4. RÉCUPÉRER LE COURS COMPLET AVEC SES CHAPITRES ET LEÇONS
        // ============================================================
        Cours cours = coursDAO.trouverParId(coursId);
        if (cours == null) {
            session.setAttribute("messageErreur", "Ce cours n'existe pas.");
            response.sendRedirect(request.getContextPath() + "/joueur/cours");
            return;
        }

        // Récupérer tous les chapitres avec leurs leçons
        List<Chapitre> chapitres = chapitreDAO.listerParCours(coursId);
        cours.setChapitres(chapitres);
        cours.setNombreChapitres(chapitres.size());
        cours.setNombreLecons(cours.getNombreTotalLecons());

        // ============================================================
        // 5. VÉRIFIER SI L'UTILISATEUR A DÉJÀ UN CERTIFICAT
        // ============================================================
        Certificat certificatExistant = certificatDAO.trouverParUtilisateurEtCours(utilisateurId, coursId);

        // ============================================================
        // 6. ORGANISER LES CHAPITRES PAR MODULE
        // ============================================================
        Map<Integer, List<Chapitre>> chapitresParModule = new LinkedHashMap<>();
        for (Chapitre chapitre : chapitres) {
            int moduleNumero = chapitre.getModuleNumero() > 0 ? chapitre.getModuleNumero() : 1;
            if (!chapitresParModule.containsKey(moduleNumero)) {
                chapitresParModule.put(moduleNumero, new ArrayList<>());
            }
            chapitresParModule.get(moduleNumero).add(chapitre);
        }

        // ============================================================
        // 7. RÉCUPÉRER TOUS LES EXAMENS DU COURS
        // ============================================================
        List<ExamenModule> examens = examenModuleDAO.listerParCours(coursId);

        // ============================================================
        // 8. CONSTRUIRE LES DONNÉES DES MODULES AVEC PROGRESSION
        // ============================================================
        List<Map<String, Object>> modulesData = new ArrayList<>();
        int totalLeconsCours = 0;
        int totalLeconsTermineesCours = 0;
        int totalExemplesCode = 0;
        int modulesReussis = 0;

        for (int moduleNumero = 1; moduleNumero <= NOMBRE_MODULES; moduleNumero++) {
            Map<String, Object> moduleData = new LinkedHashMap<>();
            moduleData.put("numero", moduleNumero);

            // Vérifier si le module est débloqué
            boolean debloque = progressionModuleDAO.estModuleDebloque(utilisateurId, coursId, moduleNumero);
            moduleData.put("debloque", debloque);

            // Récupérer la progression du module
            ProgressionModule progression = progressionModuleDAO.getProgression(utilisateurId, coursId, moduleNumero);
            boolean moduleReussi = progression != null && progression.isReussi();
            moduleData.put("reussi", moduleReussi);
            moduleData.put("progression", progression);

            if (moduleReussi) {
                modulesReussis++;
            }

            // Récupérer l'examen du module
            ExamenModule examen = null;
            if (examens != null) {
                for (ExamenModule e : examens) {
                    if (e.getModuleNumero() == moduleNumero) {
                        examen = e;
                        break;
                    }
                }
            }
            moduleData.put("examen", examen);

            // Récupérer les chapitres de ce module
            List<Chapitre> chapitresModule = chapitresParModule.getOrDefault(moduleNumero, new ArrayList<>());
            moduleData.put("nbChapitres", chapitresModule.size());

            // Construire les données des chapitres avec progression
            List<Map<String, Object>> chapitresData = new ArrayList<>();
            int leconsModule = 0;
            int leconsTermineesModule = 0;
            int chapitresTerminesModule = 0;

            // Trouver la première leçon du module (pour le bouton "Commencer")
            int premiereLeconId = 0;
            int premierChapitreId = 0;

            for (Chapitre chapitre : chapitresModule) {
                Map<String, Object> chapitreData = new LinkedHashMap<>();
                chapitreData.put("chapitre", chapitre);

                boolean chapitreTermine = progressionCoursDAO.estChapitreTermine(utilisateurId, chapitre.getId());
                chapitreData.put("termine", chapitreTermine);
                if (chapitreTermine) chapitresTerminesModule++;

                // Leçons du chapitre avec progression
                List<Lecon> lecons = leconDAO.listerParChapitre(chapitre.getId());
                if (lecons == null) lecons = new ArrayList<>();
                chapitre.setLecons(lecons);

                List<Map<String, Object>> leconsData = new ArrayList<>();
                int leconsTermineesChapitre = 0;

                for (int i = 0; i < lecons.size(); i++) {
                    Lecon lecon = lecons.get(i);
                    Map<String, Object> leconData = new LinkedHashMap<>();
                    leconData.put("lecon", lecon);
                    leconData.put("ordre", i + 1);

                    boolean leconTerminee = progressionCoursDAO.estLeconTerminee(utilisateurId, lecon.getId());
                    leconData.put("terminee", leconTerminee);
                    if (leconTerminee) leconsTermineesChapitre++;

                    // Si le module est débloqué ou réussi, toutes les leçons sont accessibles
                    boolean leconAccessible = debloque || moduleReussi;
                    leconData.put("accessible", leconAccessible);

                    // Compter les exemples de code
                    if (lecon.hasCodeSource()) {
                        totalExemplesCode++;
                    }

                    // Mémoriser la première leçon accessible du module
                    if (leconAccessible && premiereLeconId == 0) {
                        premiereLeconId = lecon.getId();
                        premierChapitreId = chapitre.getId();
                    }

                    leconsData.add(leconData);
                }

                chapitreData.put("lecons", leconsData);
                chapitreData.put("nbLecons", lecons.size());
                chapitreData.put("leconsTerminees", leconsTermineesChapitre);
                chapitreData.put("pourcentage", lecons.size() > 0 ? 
                    (leconsTermineesChapitre * 100 / lecons.size()) : 0);

                chapitresData.add(chapitreData);

                leconsModule += lecons.size();
                leconsTermineesModule += leconsTermineesChapitre;
            }

            moduleData.put("chapitres", chapitresData);
            moduleData.put("nbChapitres", chapitresModule.size());
            moduleData.put("chapitresTermines", chapitresTerminesModule);
            moduleData.put("nbLecons", leconsModule);
            moduleData.put("leconsTerminees", leconsTermineesModule);
            moduleData.put("pourcentage", leconsModule > 0 ? 
                (leconsTermineesModule * 100 / leconsModule) : 0);
            moduleData.put("premiereLeconId", premiereLeconId);
            moduleData.put("premierChapitreId", premierChapitreId);

            modulesData.add(moduleData);

            totalLeconsCours += leconsModule;
            totalLeconsTermineesCours += leconsTermineesModule;
        }

        // ============================================================
        // 9. CALCULER LES STATISTIQUES GLOBALES DU COURS
        // ============================================================
        boolean coursTermine = progressionCoursDAO.estCoursTermine(utilisateurId, coursId);
        int pourcentageGlobal = (totalLeconsCours > 0) ? 
            (totalLeconsTermineesCours * 100 / totalLeconsCours) : 0;

        // Calculer le temps de lecture total estimé
        int tempsLectureTotalMinutes = 0;
        for (Chapitre chapitre : chapitres) {
            List<Lecon> leconsChap = leconDAO.listerParChapitre(chapitre.getId());
            if (leconsChap != null) {
                for (Lecon lecon : leconsChap) {
                    tempsLectureTotalMinutes += lecon.getTempsLectureEstime();
                }
            }
        }
        String tempsLectureTotalFormate = formaterTempsLecture(tempsLectureTotalMinutes);

        // ============================================================
        // 10. DÉTERMINER SI L'EXAMEN FINAL EST ACCESSIBLE
        // ============================================================
        boolean examenAccessible = modulesReussis >= NOMBRE_MODULES && certificatExistant == null;
        boolean dejaCertifie = certificatExistant != null;

        // ============================================================
        // 11. DÉTERMINER LE PROCHAIN MODULE À FAIRE
        // ============================================================
        Integer prochainModule = null;
        for (int m = 1; m <= NOMBRE_MODULES; m++) {
            if (progressionModuleDAO.estModuleDebloque(utilisateurId, coursId, m) && 
                !progressionModuleDAO.estModuleReussi(utilisateurId, coursId, m)) {
                prochainModule = m;
                break;
            }
        }

        // ============================================================
        // 12. MESSAGE DE MOTIVATION
        // ============================================================
        String messageMotivation;
        String iconeMotivation;
        if (dejaCertifie) {
            messageMotivation = "Félicitations ! Vous avez déjà obtenu le certificat pour ce cours.";
            iconeMotivation = "fas fa-crown";
        } else if (modulesReussis >= NOMBRE_MODULES) {
            messageMotivation = "Tous les modules sont terminés ! Passez l'examen final pour obtenir votre certificat.";
            iconeMotivation = "fas fa-trophy";
        } else if (modulesReussis >= NOMBRE_MODULES / 2) {
            messageMotivation = "Vous êtes à mi-chemin ! Continuez vos efforts pour terminer ce cours.";
            iconeMotivation = "fas fa-fire";
        } else if (modulesReussis > 0) {
            messageMotivation = "Bon début ! Chaque module réussi vous rapproche de la maîtrise de ce cours.";
            iconeMotivation = "fas fa-thumbs-up";
        } else {
            messageMotivation = "Commencez dès maintenant votre apprentissage par la lecture et la pratique !";
            iconeMotivation = "fas fa-rocket";
        }

        // ============================================================
        // 13. PASSAGE DES ATTRIBUTS À LA JSP
        // ============================================================
        request.setAttribute("pageTitle", "Plan - " + cours.getTitre() + " - AmijayQuiz");
        request.setAttribute("cours", cours);
        request.setAttribute("modules", modulesData);
        request.setAttribute("nombreModules", NOMBRE_MODULES);
        request.setAttribute("modulesReussis", modulesReussis);
        request.setAttribute("totalChapitres", chapitres.size());
        request.setAttribute("totalLecons", totalLeconsCours);
        request.setAttribute("totalLeconsTerminees", totalLeconsTermineesCours);
        request.setAttribute("totalExemplesCode", totalExemplesCode);
        request.setAttribute("tempsLectureTotal", tempsLectureTotalFormate);
        request.setAttribute("pourcentageGlobal", pourcentageGlobal);
        request.setAttribute("coursTermine", coursTermine);
        request.setAttribute("examenAccessible", examenAccessible);
        request.setAttribute("dejaCertifie", dejaCertifie);
        request.setAttribute("certificatExistant", certificatExistant);
        request.setAttribute("prochainModule", prochainModule);
        request.setAttribute("messageMotivation", messageMotivation);
        request.setAttribute("iconeMotivation", iconeMotivation);

        // ============================================================
        // 14. AFFICHAGE DE LA PAGE
        // ============================================================
        request.getRequestDispatcher("/views/joueur/cours-plan.jsp").forward(request, response);
    }

    /**
     * Formate un temps en minutes en chaîne lisible
     * Exemples : 45 → "45 min", 120 → "2h", 150 → "2h30"
     * 
     * @param minutes Temps en minutes
     * @return Chaîne formatée
     */
    private String formaterTempsLecture(int minutes) {
        if (minutes == 0) {
            return "Moins d'1 min";
        }
        if (minutes < 60) {
            return minutes + " min";
        }
        int heures = minutes / 60;
        int minsRestantes = minutes % 60;
        if (minsRestantes == 0) {
            return heures + "h";
        }
        return heures + "h" + String.format("%02d", minsRestantes);
    }

    @Override
    protected void doPost(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        doGet(request, response);
    }
}

 * LeconCoursServlet - Page d'une leçon de cours

 * Charge et affiche le contenu complet d'une leçon :
 * - Texte enrichi (HTML)
 * - Code source de démonstration

 * URL : /joueur/cours/lecon?coursId=1&chapitreId=1&leconId=1
 * 
 * @author AmijayQuiz
 * @version 3.0
 */
@WebServlet(name = "LeconCoursServlet", urlPatterns = {"/joueur/cours/lecon"})
public class LeconCoursServlet extends HttpServlet {

    // ===== CONSTANTES =====
    private static final int NOMBRE_MODULES = 15;

    // ============================================================
    // MÉTHODE DOGET - AFFICHAGE DE LA LEÇON
    // ============================================================
    @Override
    protected void doGet(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {

        // ----------------------------------------------------------
        // 1. VÉRIFICATION DE LA SESSION ET DU RÔLE
        // ----------------------------------------------------------
        HttpSession session = request.getSession(false);
        if (session == null || session.getAttribute("utilisateur") == null) {
            response.sendRedirect(request.getContextPath() + "/connexion");
            return;
        }

        Utilisateur utilisateur = (Utilisateur) session.getAttribute("utilisateur");
        if (!"joueur".equals(utilisateur.getRole())) {
            response.sendRedirect(request.getContextPath() + "/connexion");
            return;
        }

        int utilisateurId = utilisateur.getId();

        // ----------------------------------------------------------
        // 2. RÉCUPÉRATION ET VALIDATION DES PARAMÈTRES
        // ----------------------------------------------------------
        String coursIdParam = request.getParameter("coursId");
        String chapitreIdParam = request.getParameter("chapitreId");
        String leconIdParam = request.getParameter("leconId");

        if (coursIdParam == null || coursIdParam.trim().isEmpty() ||
            chapitreIdParam == null || chapitreIdParam.trim().isEmpty() ||
            leconIdParam == null || leconIdParam.trim().isEmpty()) {
            response.sendRedirect(request.getContextPath() + "/joueur/cours");
            return;
        }

        int coursId;
        int chapitreId;
        int leconId;

        try {
            coursId = Integer.parseInt(coursIdParam.trim());
            chapitreId = Integer.parseInt(chapitreIdParam.trim());
            leconId = Integer.parseInt(leconIdParam.trim());
        } catch (NumberFormatException e) {
            response.sendRedirect(request.getContextPath() + "/joueur/cours");
            return;
        }

        if (coursId <= 0 || chapitreId <= 0 || leconId <= 0) {
            response.sendRedirect(request.getContextPath() + "/joueur/cours");
            return;
        }

        // ----------------------------------------------------------
        // 3. INITIALISATION DES DAO
        // ----------------------------------------------------------
        CoursDAO coursDAO = new CoursDAO();
        ChapitreDAO chapitreDAO = new ChapitreDAO();
        LeconDAO leconDAO = new LeconDAO();
        ProgressionCoursDAO progressionCoursDAO = new ProgressionCoursDAO();
        ProgressionModuleDAO progressionModuleDAO = new ProgressionModuleDAO();
        ExamenModuleDAO examenModuleDAO = new ExamenModuleDAO();

        // ----------------------------------------------------------
        // 4. RÉCUPÉRATION DU COURS
        // ----------------------------------------------------------
        Cours cours = coursDAO.trouverParId(coursId);
        if (cours == null) {
            session.setAttribute("messageErreur", "Ce cours n'existe pas.");
            response.sendRedirect(request.getContextPath() + "/joueur/cours");
            return;
        }

        // ----------------------------------------------------------
        // 5. RÉCUPÉRATION DU CHAPITRE
        // ----------------------------------------------------------
        Chapitre chapitre = chapitreDAO.trouverParId(chapitreId);
        if (chapitre == null || chapitre.getCoursId() != coursId) {
            session.setAttribute("messageErreur", "Ce chapitre n'existe pas dans ce cours.");
            response.sendRedirect(request.getContextPath() + "/joueur/cours/plan?coursId=" + coursId);
            return;
        }

        // ----------------------------------------------------------
        // 6. RÉCUPÉRATION DE LA LEÇON
        // ----------------------------------------------------------
        Lecon lecon = leconDAO.trouverParId(leconId);
        if (lecon == null || lecon.getChapitreId() != chapitreId) {
            session.setAttribute("messageErreur", "Cette leçon n'existe pas dans ce chapitre.");
            response.sendRedirect(request.getContextPath() + "/joueur/cours/chapitre?coursId=" +
                    coursId + "&chapitreId=" + chapitreId);
            return;
        }

        // ----------------------------------------------------------
        // 7. DÉTERMINER LE MODULE ACTIF
        // ----------------------------------------------------------
        int moduleActifNumero = chapitre.getModuleNumero();
        if (moduleActifNumero <= 0) {
            moduleActifNumero = 1;
        }

        // ----------------------------------------------------------
        // 8. VÉRIFIER SI LE MODULE EST DÉBLOQUÉ OU RÉUSSI
        // ----------------------------------------------------------
        boolean moduleDebloque = progressionModuleDAO.estModuleDebloque(utilisateurId, coursId, moduleActifNumero);
        boolean moduleReussi = progressionModuleDAO.estModuleReussi(utilisateurId, coursId, moduleActifNumero);

        if (!moduleDebloque && !moduleReussi) {
            session.setAttribute("messageErreur",
                    "Ce module n'est pas encore débloqué. Réussissez d'abord le module " +
                    (moduleActifNumero - 1) + ".");
            response.sendRedirect(request.getContextPath() + "/joueur/cours/plan?coursId=" + coursId);
            return;
        }

        // ----------------------------------------------------------
        // 9. ENREGISTRER LE DÉBUT DE LA LEÇON
        // ----------------------------------------------------------
        progressionCoursDAO.enregistrerDebut(utilisateurId, coursId, chapitreId, leconId);

        // ----------------------------------------------------------
        // 10. RÉCUPÉRER LA PROGRESSION ACTUELLE
        // ----------------------------------------------------------
        ProgressionCours progression = progressionCoursDAO.getProgression(utilisateurId, leconId);

        // ----------------------------------------------------------
        // 11. CHARGER TOUS LES CHAPITRES DU COURS
        // ----------------------------------------------------------
        List<Chapitre> tousLesChapitres = chapitreDAO.listerParCours(coursId);
        cours.setChapitres(tousLesChapitres);

        // ----------------------------------------------------------
        // 12. FILTRER LES CHAPITRES DU MODULE ACTIF
        // ----------------------------------------------------------
        List<Chapitre> chapitresDuModule = new ArrayList<>();
        for (Chapitre chap : tousLesChapitres) {
            int chapModule = chap.getModuleNumero();
            if (chapModule <= 0) {
                chapModule = 1;
            }
            if (chapModule == moduleActifNumero) {
                chapitresDuModule.add(chap);
            }
        }

        // ----------------------------------------------------------
        // 13. CONSTRUIRE LA LISTE DE TOUTES LES LEÇONS DU MODULE
        // ----------------------------------------------------------
        List<Lecon> toutesLesLeconsDuModule = new ArrayList<>();
        for (Chapitre chap : chapitresDuModule) {
            List<Lecon> leconsChap = leconDAO.listerParChapitre(chap.getId());
            if (leconsChap != null) {
                toutesLesLeconsDuModule.addAll(leconsChap);
            }
        }

        // ----------------------------------------------------------
        // 14. TROUVER LA LEÇON PRÉCÉDENTE ET SUIVANTE DANS LE MODULE
        // ----------------------------------------------------------
        Lecon leconPrecedente = null;
        Lecon leconSuivante = null;

        for (int i = 0; i < toutesLesLeconsDuModule.size(); i++) {
            Lecon leconCourante = toutesLesLeconsDuModule.get(i);
            if (leconCourante.getId() == leconId) {
                if (i > 0) {
                    leconPrecedente = toutesLesLeconsDuModule.get(i - 1);
                }
                if (i < toutesLesLeconsDuModule.size() - 1) {
                    leconSuivante = toutesLesLeconsDuModule.get(i + 1);
                }
                break;
            }
        }

        // ----------------------------------------------------------
        // 15. DÉTERMINER SI C'EST LA DERNIÈRE LEÇON DU MODULE
        // ----------------------------------------------------------
        boolean estDerniereLeconDuModule = (leconSuivante == null);

        // ----------------------------------------------------------
        // 16. VÉRIFIER SI TOUS LES MODULES SONT RÉUSSIS
        // ----------------------------------------------------------
        boolean tousModulesReussis = true;
        for (int m = 1; m <= NOMBRE_MODULES; m++) {
            if (!progressionModuleDAO.estModuleReussi(utilisateurId, coursId, m)) {
                tousModulesReussis = false;
                break;
            }
        }

        // ----------------------------------------------------------
        // 17. RÉCUPÉRER TOUS LES EXAMENS DU COURS
        // ----------------------------------------------------------
        List<ExamenModule> examens = examenModuleDAO.listerParCours(coursId);

        // ----------------------------------------------------------
        // 18. CONSTRUIRE LES DONNÉES DES MODULES POUR LA BARRE LATÉRALE
        // ----------------------------------------------------------
        List<Map<String, Object>> modulesData = new ArrayList<>();

        for (int moduleNumero = 1; moduleNumero <= NOMBRE_MODULES; moduleNumero++) {

            Map<String, Object> moduleData = new LinkedHashMap<>();
            moduleData.put("numero", moduleNumero);

            // Statut du module
            boolean debloque = progressionModuleDAO.estModuleDebloque(utilisateurId, coursId, moduleNumero);
            boolean reussi = progressionModuleDAO.estModuleReussi(utilisateurId, coursId, moduleNumero);
            moduleData.put("debloque", debloque);
            moduleData.put("reussi", reussi);

            // Progression du module
            ProgressionModule progressionModule = progressionModuleDAO.getProgression(
                    utilisateurId, coursId, moduleNumero);
            moduleData.put("progression", progressionModule);

            // Examen du module
            ExamenModule examen = null;
            if (examens != null) {
                for (ExamenModule e : examens) {
                    if (e.getModuleNumero() == moduleNumero) {
                        examen = e;
                        break;
                    }
                }
            }
            moduleData.put("examen", examen);

            // Filtrer les chapitres de ce module
            List<Chapitre> chapitresModule = new ArrayList<>();
            for (Chapitre chap : tousLesChapitres) {
                int chapModule = chap.getModuleNumero();
                if (chapModule <= 0) {
                    chapModule = 1;
                }
                if (chapModule == moduleNumero) {
                    chapitresModule.add(chap);
                }
            }

            int nbChapitresModule = chapitresModule.size();
            int nbLeconsModule = 0;

            List<Map<String, Object>> chapitresData = new ArrayList<>();

            for (Chapitre chap : chapitresModule) {

                Map<String, Object> chapitreData = new LinkedHashMap<>();
                chapitreData.put("chapitre", chap);

                boolean chapTermine = progressionCoursDAO.estChapitreTermine(utilisateurId, chap.getId());
                chapitreData.put("termine", chapTermine);

                List<Lecon> leconsChap = leconDAO.listerParChapitre(chap.getId());
                if (leconsChap == null) {
                    leconsChap = new ArrayList<>();
                }

                List<Map<String, Object>> leconsData = new ArrayList<>();

                for (int i = 0; i < leconsChap.size(); i++) {
                    Lecon leconDuChap = leconsChap.get(i);

                    Map<String, Object> leconData = new LinkedHashMap<>();
                    leconData.put("lecon", leconDuChap);
                    leconData.put("ordre", i + 1);

                    boolean leconTerminee = progressionCoursDAO.estLeconTerminee(
                            utilisateurId, leconDuChap.getId());
                    leconData.put("terminee", leconTerminee);

                    boolean leconAccessible = (debloque || reussi);
                    leconData.put("accessible", leconAccessible);

                    leconsData.add(leconData);
                }

                chapitreData.put("lecons", leconsData);
                chapitreData.put("nbLecons", leconsChap.size());

                chapitresData.add(chapitreData);

                nbLeconsModule += leconsChap.size();
            }

            moduleData.put("chapitres", chapitresData);
            moduleData.put("nbChapitres", nbChapitresModule);
            moduleData.put("nbLecons", nbLeconsModule);

            modulesData.add(moduleData);
        }

        // ----------------------------------------------------------
        // 19. PASSAGE DES ATTRIBUTS À LA JSP
        // ----------------------------------------------------------
        request.setAttribute("pageTitle", lecon.getTitre() + " - " + cours.getTitre() + " - AmijayQuiz");
        request.setAttribute("cours", cours);
        request.setAttribute("chapitre", chapitre);
        request.setAttribute("lecon", lecon);
        request.setAttribute("progression", progression);
        request.setAttribute("leconPrecedente", leconPrecedente);
        request.setAttribute("leconSuivante", leconSuivante);
        request.setAttribute("estDerniereLecon", estDerniereLeconDuModule);
        request.setAttribute("moduleDebloque", moduleDebloque);
        request.setAttribute("moduleReussi", moduleReussi);
        request.setAttribute("moduleSuivant", moduleActifNumero < NOMBRE_MODULES ? moduleActifNumero + 1 : 0);
        request.setAttribute("tousChapitresTermines", tousModulesReussis);
        request.setAttribute("totalLeconsChapitre", leconDAO.compterParChapitre(chapitreId));
        request.setAttribute("modules", modulesData);
        request.setAttribute("moduleActiveNumero", moduleActifNumero);
        request.setAttribute("leconActiveId", lecon.getId());
        request.setAttribute("chapitreActiveId", chapitre.getId());

        // ----------------------------------------------------------
        // 20. AFFICHAGE DE LA PAGE
        // ----------------------------------------------------------
        request.getRequestDispatcher("/views/joueur/cours-lecon.jsp").forward(request, response);
    }

    // ============================================================
    // MÉTHODE DOPOST - REDIRIGE VERS DOGET
    // ============================================================
    @Override
    protected void doPost(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        doGet(request, response);
    }
}

import javax.servlet.http.HttpServlet;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;
import javax.servlet.http.HttpSession;
import java.io.IOException;
import java.util.ArrayList;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;

/**
 * PlanCoursServlet - Plan détaillé d'un cours
 * 
 * Affiche l'arborescence complète d'un cours :
 * - Tous les chapitres avec leurs titres et descriptions
 * - Toutes les leçons de chaque chapitre avec leur statut
 * - Progression globale du cours
 * - Statut de chaque chapitre (terminé, en cours, verrouillé)
 * - Accès à l'examen final si tous les chapitres sont terminés
 * - Lien vers le certificat si déjà obtenu
 * 
 * URL : /joueur/cours/plan?coursId=1
 * 
 * @author AmijayQuiz
 * @version 1.1
 */
@WebServlet(name = "PlanCoursServlet", urlPatterns = {"/joueur/cours/plan"})
public class PlanCoursServlet extends HttpServlet {

    @Override
    protected void doGet(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {

        // 1. Vérification de la session et du rôle
        HttpSession session = request.getSession(false);
        if (session == null || session.getAttribute("utilisateur") == null) {
            response.sendRedirect(request.getContextPath() + "/connexion");
            return;
        }

        Utilisateur utilisateur = (Utilisateur) session.getAttribute("utilisateur");
        if (!"joueur".equals(utilisateur.getRole())) {
            response.sendRedirect(request.getContextPath() + "/connexion");
            return;
        }

        int utilisateurId = utilisateur.getId();

        // 2. Récupération et validation du paramètre coursId
        String coursIdParam = request.getParameter("coursId");
        if (coursIdParam == null || coursIdParam.trim().isEmpty()) {
            response.sendRedirect(request.getContextPath() + "/joueur/cours");
            return;
        }

        int coursId;
        try {
            coursId = Integer.parseInt(coursIdParam);
        } catch (NumberFormatException e) {
            response.sendRedirect(request.getContextPath() + "/joueur/cours");
            return;
        }

        // 3. Initialisation des DAO
        CoursDAO coursDAO = new CoursDAO();
        ChapitreDAO chapitreDAO = new ChapitreDAO();
        ProgressionCoursDAO progressionDAO = new ProgressionCoursDAO();
        CertificatDAO certificatDAO = new CertificatDAO();

        // 4. Récupérer le cours complet avec ses chapitres et leçons
        Cours cours = coursDAO.trouverParId(coursId);
        if (cours == null) {
            session.setAttribute("messageErreur", "Ce cours n'existe pas.");
            response.sendRedirect(request.getContextPath() + "/joueur/cours");
            return;
        }

        // 5. Récupérer tous les chapitres avec leurs leçons
        List<Chapitre> chapitres = chapitreDAO.listerParCours(coursId);
        cours.setChapitres(chapitres);
        cours.setNombreChapitres(chapitres.size());
        cours.setNombreLecons(cours.getNombreTotalLecons());

        // 6. Vérifier si l'utilisateur a déjà un certificat pour ce cours
        Certificat certificatExistant = certificatDAO.trouverParUtilisateurEtCours(utilisateurId, coursId);

        // 7. Construire l'arborescence avec progression
        List<Map<String, Object>> chapitresAvecProgression = new ArrayList<>();
        int totalLeconsCours = 0;
        int totalLeconsTermineesCours = 0;
        int totalExemplesCode = 0;
        int chapitresTerminesCount = 0;
        int ordreChapitre = 0;

        for (Chapitre chapitre : chapitres) {
            ordreChapitre++;
            Map<String, Object> chapitreData = new LinkedHashMap<>();
            chapitreData.put("chapitre", chapitre);
            chapitreData.put("ordre", ordreChapitre);

            // Vérifier si le chapitre est terminé
            boolean chapitreTermine = progressionDAO.estChapitreTermine(utilisateurId, chapitre.getId());
            chapitreData.put("termine", chapitreTermine);
            if (chapitreTermine) chapitresTerminesCount++;

            // Vérifier si le chapitre est accessible
            boolean estAccessible;
            if (ordreChapitre == 1) {
                estAccessible = true;
            } else {
                Chapitre chapitrePrecedent = chapitres.get(ordreChapitre - 2);
                estAccessible = progressionDAO.estChapitreTermine(utilisateurId, chapitrePrecedent.getId());
            }
            chapitreData.put("accessible", estAccessible);

            // Déterminer le statut visuel du chapitre
            String statutChapitre;
            String iconeStatutChapitre;
            String couleurStatutChapitre;
            if (chapitreTermine) {
                statutChapitre = "Terminé";
                iconeStatutChapitre = "fas fa-check-circle";
                couleurStatutChapitre = "#27ae60";
            } else if (estAccessible) {
                statutChapitre = "En cours";
                iconeStatutChapitre = "fas fa-play-circle";
                couleurStatutChapitre = "#667eea";
            } else {
                statutChapitre = "Verrouillé";
                iconeStatutChapitre = "fas fa-lock";
                couleurStatutChapitre = "#aaa";
            }
            chapitreData.put("statut", statutChapitre);
            chapitreData.put("iconeStatut", iconeStatutChapitre);
            chapitreData.put("couleurStatut", couleurStatutChapitre);

            // Leçons du chapitre avec leur progression
            List<Lecon> lecons = chapitre.getLecons();
            if (lecons == null) lecons = new ArrayList<>();
            
            List<Map<String, Object>> leconsAvecProgression = new ArrayList<>();
            int leconsTermineesChapitre = 0;

            for (int i = 0; i < lecons.size(); i++) {
                Lecon lecon = lecons.get(i);
                Map<String, Object> leconData = new LinkedHashMap<>();
                leconData.put("lecon", lecon);
                leconData.put("ordre", i + 1);

                boolean leconTerminee = progressionDAO.estLeconTerminee(utilisateurId, lecon.getId());
                leconData.put("terminee", leconTerminee);
                if (leconTerminee) leconsTermineesChapitre++;

                // Accessibilité de la leçon dans le chapitre
                boolean leconAccessible;
                if (i == 0) {
                    leconAccessible = estAccessible;
                } else {
                    leconAccessible = estAccessible && 
                        progressionDAO.estLeconTerminee(utilisateurId, lecons.get(i - 1).getId());
                }
                leconData.put("accessible", leconAccessible);

                // Compter les exemples de code
                if (lecon.hasCodeSource()) {
                    totalExemplesCode++;
                }

                leconsAvecProgression.add(leconData);
            }

            chapitreData.put("lecons", leconsAvecProgression);
            chapitreData.put("nbLecons", lecons.size());
            chapitreData.put("leconsTerminees", leconsTermineesChapitre);
            chapitreData.put("pourcentageChapitre", lecons.size() > 0 ? 
                (leconsTermineesChapitre * 100 / lecons.size()) : 0);

            chapitresAvecProgression.add(chapitreData);

            totalLeconsCours += lecons.size();
            totalLeconsTermineesCours += leconsTermineesChapitre;
        }

        // 8. Calculer les statistiques globales du cours
        boolean coursTermine = progressionDAO.estCoursTermine(utilisateurId, coursId);
        int pourcentageGlobal = (totalLeconsCours > 0) ? 
            (totalLeconsTermineesCours * 100 / totalLeconsCours) : 0;

        // 9. Calculer le temps de lecture total estimé
        int tempsLectureTotalMinutes = 0;
        for (Chapitre chapitre : chapitres) {
            if (chapitre.getLecons() != null) {
                for (Lecon lecon : chapitre.getLecons()) {
                    tempsLectureTotalMinutes += lecon.getTempsLectureEstime();
                }
            }
        }
        String tempsLectureTotalFormate = formaterTempsLecture(tempsLectureTotalMinutes);

        // 10. Déterminer si l'examen final est accessible
        boolean examenAccessible = coursTermine && certificatExistant == null;
        boolean dejaCertifie = certificatExistant != null;

        // 11. Déterminer le prochain chapitre à faire
        Integer prochainChapitreId = null;
        String prochainChapitreTitre = null;
        for (Map<String, Object> chapData : chapitresAvecProgression) {
            boolean termine = (Boolean) chapData.get("termine");
            boolean accessible = (Boolean) chapData.get("accessible");
            if (!termine && accessible && prochainChapitreId == null) {
                Chapitre chap = (Chapitre) chapData.get("chapitre");
                prochainChapitreId = chap.getId();
                prochainChapitreTitre = chap.getTitre();
                break;
            }
        }

        // 12. Message de motivation
        String messageMotivation;
        String iconeMotivation;
        if (dejaCertifie) {
            messageMotivation = "Félicitations ! Vous avez déjà obtenu le certificat pour ce cours.";
            iconeMotivation = "fas fa-crown";
        } else if (coursTermine) {
            messageMotivation = "Tous les chapitres sont terminés ! Passez l'examen final pour obtenir votre certificat.";
            iconeMotivation = "fas fa-trophy";
        } else if (pourcentageGlobal >= 50) {
            messageMotivation = "Vous êtes à mi-chemin ! Continuez vos efforts pour terminer ce cours.";
            iconeMotivation = "fas fa-fire";
        } else if (pourcentageGlobal > 0) {
            messageMotivation = "Bon début ! Chaque leçon vous rapproche de la maîtrise de ce cours.";
            iconeMotivation = "fas fa-thumbs-up";
        } else {
            messageMotivation = "Commencez dès maintenant votre apprentissage par la lecture et la pratique !";
            iconeMotivation = "fas fa-rocket";
        }

        // 13. Passage des attributs à la JSP
        request.setAttribute("pageTitle", "Plan - " + cours.getTitre() + " - AmijayQuiz");
        request.setAttribute("cours", cours);
        request.setAttribute("chapitresAvecProgression", chapitresAvecProgression);
        request.setAttribute("totalChapitres", chapitres.size());
        request.setAttribute("chapitresTermines", chapitresTerminesCount);
        request.setAttribute("totalLecons", totalLeconsCours);
        request.setAttribute("totalLeconsTerminees", totalLeconsTermineesCours);
        request.setAttribute("totalExemplesCode", totalExemplesCode);
        request.setAttribute("tempsLectureTotal", tempsLectureTotalFormate);
        request.setAttribute("pourcentageGlobal", pourcentageGlobal);
        request.setAttribute("coursTermine", coursTermine);
        request.setAttribute("examenAccessible", examenAccessible);
        request.setAttribute("dejaCertifie", dejaCertifie);
        request.setAttribute("certificatExistant", certificatExistant);
        request.setAttribute("prochainChapitreId", prochainChapitreId);
        request.setAttribute("prochainChapitreTitre", prochainChapitreTitre);
        request.setAttribute("messageMotivation", messageMotivation);
        request.setAttribute("iconeMotivation", iconeMotivation);

        // 14. Affichage de la page
        request.getRequestDispatcher("/views/joueur/cours-plan.jsp").forward(request, response);
    }

    /**
     * Formate un temps en minutes en chaîne lisible
     * Exemples : 45 → "45 min", 120 → "2h", 150 → "2h30"
     * 
     * @param minutes Temps en minutes
     * @return Chaîne formatée
     */
    private String formaterTempsLecture(int minutes) {
        if (minutes == 0) {
            return "Moins d'1 min";
        }
        if (minutes < 60) {
            return minutes + " min";
        }
        int heures = minutes / 60;
        int minsRestantes = minutes % 60;
        if (minsRestantes == 0) {
            return heures + "h";
        }
        return heures + "h" + String.format("%02d", minsRestantes);
    }

    @Override
    protected void doPost(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        doGet(request, response);
    }
}
